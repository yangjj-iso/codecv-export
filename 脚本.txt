// ==UserScript==
// @name         CodeCV 简历打印为无水印 PDF
// @namespace    local.codecv.print
// @version      1.1.0
// @description  将当前简历预览复制到独立打印页，保存为 PDF。
// @match        https://www.codecvcv.com/cv/*
// @match        https://codecvcv.com/cv/*
// @sandbox      DOM
// @run-at       document-idle
// ==/UserScript==

(() => {
  'use strict';

  const BUTTON_ID = 'codecv-local-print-button';
  if (document.getElementById(BUTTON_ID)) return;

  const button = document.createElement('button');
  button.id = BUTTON_ID;
  button.type = 'button';
  button.textContent = '打印简历 PDF';
  Object.assign(button.style, {
    position: 'fixed',
    right: '16px',
    bottom: '16px',
    zIndex: '2147483647',
    padding: '10px 14px',
    border: '0',
    borderRadius: '8px',
    background: '#24292f',
    color: '#fff',
    cursor: 'pointer',
    boxShadow: '0 2px 10px #0003',
    font: '14px/1.4 sans-serif',
  });

  button.addEventListener('click', async () => {
    const preview = document.querySelector('.preview-cv .re-render')
      || document.querySelector('.re-render');
    if (!preview) {
      window.alert('没有找到简历预览。请先打开 CodeCV 的简历编辑页，并等待预览加载。');
      return;
    }

    button.disabled = true;
    button.textContent = '准备打印…';
    let frame;
    try {
      frame = document.createElement('iframe');
      frame.setAttribute('aria-hidden', 'true');
      Object.assign(frame.style, {
        position: 'fixed',
        left: '-10000px',
        top: '0',
        width: '794px',
        height: '1123px',
        border: '0',
        opacity: '0',
        pointerEvents: 'none',
      });
      document.body.appendChild(frame);

      const printDoc = frame.contentDocument;
      const base = printDoc.createElement('base');
      base.href = location.href;
      printDoc.head.appendChild(base);
      printDoc.title = document.title || 'CodeCV 简历';
      printDoc.documentElement.className = document.documentElement.className;
      printDoc.body.className = document.body.className;

      // 保留网站模板的原有样式，但不复制脚本或编辑器界面。
      const stylesheetLoads = [];
      document.querySelectorAll('link[rel="stylesheet"], style').forEach((node) => {
        const clone = node.cloneNode(true);
        if (clone.tagName === 'LINK') {
          stylesheetLoads.push(new Promise((resolve) => {
            clone.addEventListener('load', resolve, { once: true });
            clone.addEventListener('error', resolve, { once: true });
          }));
        }
        printDoc.head.appendChild(clone);
      });

      const copy = printDoc.importNode(preview, true);
      copy.querySelectorAll('*').forEach((node) => {
        const marker = `${node.id || ''} ${node.className?.baseVal || node.className || ''}`;
        if (/watermark/i.test(marker)) node.remove();
      });
      printDoc.body.appendChild(copy);

      const printStyle = printDoc.createElement('style');
      printStyle.textContent = `
        @page { size: A4 portrait; margin: 0; }
        html, body { margin: 0 !important; padding: 0 !important;
          width: 210mm !important; height: auto !important;
          overflow: visible !important; background: white !important; }
        .re-render { display: block !important; position: relative !important;
          margin: 0 !important; transform: none !important;
          box-shadow: none !important; }
        [class*="watermark" i], [id*="watermark" i] { display: none !important; }
        .preview-cv::before, .preview-cv::after { content: none !important; }
        * { -webkit-print-color-adjust: exact !important;
          print-color-adjust: exact !important; }
      `;
      printDoc.head.appendChild(printStyle);

      const ready = async () => {
        await Promise.all(stylesheetLoads);
        await Promise.all([
          printDoc.fonts?.ready || Promise.resolve(),
          ...Array.from(printDoc.images, (img) => img.complete
          ? Promise.resolve()
          : new Promise((resolve) => {
            img.addEventListener('load', resolve, { once: true });
            img.addEventListener('error', resolve, { once: true });
          })),
        ]);
      };
      await Promise.race([
        ready(),
        new Promise((resolve) => window.setTimeout(resolve, 5_000)),
      ]);

      // 浏览器可能在打印对话框关闭后才触发 afterprint。
      frame.contentWindow.addEventListener('afterprint', () => frame.remove(), { once: true });
      frame.contentWindow.focus();
      frame.contentWindow.print();
      window.setTimeout(() => frame?.remove(), 60_000);
    } catch (error) {
      frame?.remove();
      console.error('[CodeCV print]', error);
      window.alert('打印失败，请打开开发者工具查看错误信息。');
    } finally {
      button.disabled = false;
      button.textContent = '打印简历 PDF';
    }
  });

  const mountButton = () => {
    if (!document.body || document.getElementById(BUTTON_ID)) return;
    document.body.appendChild(button);
  };

  if (document.body) mountButton();
  else document.addEventListener('DOMContentLoaded', mountButton, { once: true });

  // CodeCV 是单页应用；页面重建 body 时重新挂载按钮。
  const observer = new MutationObserver(() => {
    if (!button.isConnected) mountButton();
  });
  observer.observe(document.documentElement, { childList: true });
  if (document.body) observer.observe(document.body, { childList: true });
})();
